# ===== CONTEXT =====
# redis-features-01.prod.internal: single node, 32 GB RAM, no swap. Redis 7, systemd unit redis-server.
# maxmemory 24gb, maxmemory-policy noeviction, stop-writes-on-bgsave-error yes.
# fraud-scorer writes per-card velocity counters (INCR + TTL) on every score.
# Rejected write -> fraud-scorer logs FRD-E140; velocity counters go stale until writes recover.

# ===== SETUP =====
r() { redis-cli -h redis-features-01.prod.internal "$@"; }   # function, so it works in bash and zsh

# ===== IS IT WRITES? =====
kubectl -n risk logs -l app=fraud-scorer --prefix --tail=-1 --since=30m | grep FRD-E140 | awk '{print $1}' | sort | uniq -c   # per pod
r PING                     # PONG = server is up, reads work
r INCR oncall:probe      # MISCONF... or OOM... = writes refused
r DEL oncall:probe

# ===== WHICH ONE =====
r INFO persistence | grep -E '^(rdb_last_bgsave_status|rdb_bgsave_in_progress|rdb_changes_since_last_save):'
r INFO memory | grep -E '^(used_memory_human|maxmemory_human|maxmemory_policy):'
# rdb_last_bgsave_status:err                -> MISCONF path
# used_memory at maxmemory, noeviction       -> OOM path ("OOM command not allowed when used memory > 'maxmemory'")

# ===== MISCONF PATH: WHY DID BGSAVE FAIL =====
ssh redis-features-01.prod.internal 'sudo journalctl -u redis-server --since "-2h" | grep -iE "background saving|fork|MISCONF"'
#   "fork: Cannot allocate memory" -> commit limit
ssh redis-features-01.prod.internal 'sysctl vm.overcommit_memory vm.overcommit_ratio; grep -E "^(CommitLimit|Committed_AS):" /proc/meminfo'
#   expected: vm.overcommit_memory = 1
#   with 2 (strict), CommitLimit = swap + RAM * overcommit_ratio%. The fork's child is
#   charged about as much as Redis already uses, so once Redis holds about half the
#   limit, BGSAVE's fork fails and stop-writes-on-bgsave-error refuses writes (MISCONF).
#   disk full on the RDB dir also fails BGSAVE:
ssh redis-features-01.prod.internal 'df -h /var/lib/redis'

# ===== MISCONF PATH: FIX =====
ssh redis-features-01.prod.internal 'sudo sysctl -w vm.overcommit_memory=1'      # then fix it at the source (baseline / sysctl.d)
r BGSAVE
r INFO persistence | grep rdb_last_bgsave_status   # ok -> writes accepted again
# do NOT set stop-writes-on-bgsave-error no to get writes back; you lose the signal that snapshots are failing

# ===== OOM PATH =====
r INFO keyspace        # keys vs expires: counters carry a TTL, features don't
r --memkeys --memkeys-samples 0 -i 0.1    # SCANs the keyspace; off-peak, it adds latency
# no eviction by design: feature keys must not vanish silently. free memory by deleting what you found,
# or raise maxmemory only with headroom for the BGSAVE fork. page team-risk before deleting feature keys.

# ===== DONE =====
# INCR probe succeeds, FRD-E140 back to 0, rdb_last_bgsave_status:ok, used_memory under maxmemory

# ===== HISTORY =====
# 2025-09: team-platform's OS hardening baseline set vm.overcommit_memory=2 (ratio 80) on this VM.
# Redis at ~14 GB; the fork pushed Committed_AS past the ~25.6 GB limit; "fork: Cannot allocate memory",
# MISCONF on writes, FRD-E140 from both scorer pods. Fixed by restoring 1.
